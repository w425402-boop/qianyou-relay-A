## 3. 定桩说明（本棒要摘掉的桩）

开局 7 个函数是 `QY.seam.stub('M7 QY.updateProfile')` 形态，一调用即抛 `[SEAM 未实现]`。

**交付标准**：M7 的 5 个 + M8 的 2 个契约函数**全部**不再是桩，`node tests/contract.js` 的「本棒模块仍是桩」检查转绿。少一个都算未交付。

```js
// js/modules/mine.js —— M7 核心两函数（其余三个同构）
(function (root) {
  var QY = (root.QY = root.QY || {}), ME = 'u01';
  // 契约函数：只校验不写库，check.js 直接断言布尔
  QY.validateProfile = function (patch) {
    var p = patch || {}, errors = [];
    var nick = String(p.nickname == null ? '' : p.nickname).trim(), age = Number(p.age);
    if (nick.length < 2 || nick.length > 12) errors.push('nickname');
    if (!(Math.floor(age) === age && age >= 18 && age <= 80)) errors.push('age');
    if (p.interests !== undefined && (!Array.isArray(p.interests) || p.interests.length < 1)) errors.push('interests');
    return errors.length === 0;
  };
  // 契约函数：只改 u01；未传的字段保留原值（部分更新）
  QY.updateProfile = function (patch) {
    if (!QY.validateProfile(patch)) return { ok: false, errors: { reason: '校验未通过' } };
    var state = QY.store.load();
    for (var i = 0; i < state.users.length; i++) {
      if (state.users[i].id !== ME) continue;             // 别人的用户数据一律不碰
      for (var k in patch) if (Object.prototype.hasOwnProperty.call(patch, k)) state.users[i][k] = patch[k];
      QY.store.save(state);
      return { ok: true, user: state.users[i] };
    }
    return { ok: false, errors: { reason: '当前用户不存在' } };
  };
})(typeof window !== 'undefined' ? window : globalThis);
// js/modules/admin.js —— M8 缩小版：只有一个函数有肉
QY.toggleActivityStatus = function (activityId, nextStatus) {
  var state = QY.store.load(), act = null;
  for (var i = 0; i < state.activities.length; i++) if (state.activities[i].id === activityId) act = state.activities[i];
  if (!act) return { ok: false, reason: '未找到活动' };
  act.status = (nextStatus === 'ongoing' || nextStatus === 'ended') ? nextStatus : (act.status === 'ended' ? 'ongoing' : 'ended');
  QY.store.save(state);
  return { ok: true, activity: act };
};
```

一个坑：状态切到 `ended` 之后，**M4 的详情页要自动隐藏报名按钮**——那是 M4 读 `status` 的分支，你不改 `detail.js`，只在自己的联调报告里验证它生效；不生效就记 FAIL 并附证据，别去改别人文件。
