## 3. 定桩说明（本棒要摘掉的桩）

开局 `js/seam.js` 已把这 9 个函数定成 `QY.seam.stub('M1 QY.store.load')` 形态：签名齐全、一调用就抛 `[SEAM 未实现]`。你的活就是逐个换真实现，桩工厂不许删、不许改。

**交付标准**：M1 的 9 个契约函数全部不再被 `QY.seam.isStub()` 判为桩，`node tests/contract.js` 的「本棒模块仍是桩」检查转绿。

```js
// js/store.js —— 从桩到真实现的关键骨架
(function (root) {
  var QY = (root.QY = root.QY || {});
  var KEY = 'qianyou-interview-v1';      // 契约固定键名，一个字母都不许改

  // Node 侧跑 check.js 时没有 localStorage，留内存降级分支，否则断言全红
  var mem = {};

  QY.store = {
    // 读全量状态；无数据则用 js/data.js 的种子初始化并落盘
    load: function () {
      var raw = localStorage.getItem(KEY);
      if (!raw) { var init = QY.createSeedData(); localStorage.setItem(KEY, JSON.stringify(init)); return init; }
      try { return JSON.parse(raw); }
      catch (e) { var recover = QY.createSeedData(); localStorage.setItem(KEY, JSON.stringify(recover)); return recover; }
    },
    // 写全量状态，返回落地后的对象 —— 下游断言靠这个返回值
    save: function (data) { localStorage.setItem(KEY, JSON.stringify(data)); return data; },
    // 单条查询，找不到返回 null；遍历不改入参
    getActivity: function (id) {
      var s = this.load();
      for (var i = 0; i < s.activities.length; i++) if (s.activities[i].id === id) return s.activities[i];
      return null;
    },
    getUser: function (id) {
      var s = this.load();
      for (var i = 0; i < s.users.length; i++) if (s.users[i].id === id) return s.users[i];
      return null;
    }
  };
})(typeof window !== 'undefined' ? window : globalThis);
```

两条自查：`grep -r "localStorage" js/ --exclude=store.js` 必须空；`window.QY` 在 `index.html` 里被赋值之前，`store.js` 不能假设它已存在——用 `root.QY = root.QY || {}` 兜住。
